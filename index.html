# ─── ALGORITHM DETAILS ───────────────────────
Grover Q-Learning:  Q-table + Grover amplitude amplification
                    Oracle marks best Q-action, diffusion amplifies
                    √N quantum speedup over random action selection

A* Search:          Heuristic = Manhattan distance to goal
                    Optimal path · O((V+E) log V) complexity
                    Finds shortest path guaranteed

Dijkstra:           Uniform-cost search, no heuristic
                    Also optimal · slower than A* on this maze
Core Grover circuit (Python)
import pennylane as qml
import numpy as np

dev = qml.device("default.qubit", wires=2)  # 2 qubits → 4 actions

def grover_oracle(best_action, n_qubits):
    bits = [(best_action >> i) & 1 for i in range(n_qubits)]
    for i, b in enumerate(bits):
        if b == 0: qml.PauliX(wires=i)
    qml.ctrl(qml.PauliZ, control=list(range(n_qubits-1)))(wires=n_qubits-1)
    for i, b in enumerate(bits):
        if b == 0: qml.PauliX(wires=i)

def diffusion(n_qubits):
    for w in range(n_qubits): qml.Hadamard(wires=w); qml.PauliX(wires=w)
    qml.ctrl(qml.PauliZ, control=list(range(n_qubits-1)))(wires=n_qubits-1)
    for w in range(n_qubits): qml.PauliX(wires=w); qml.Hadamard(wires=w)

@qml.qnode(dev)
def grover_circuit(best_action, n_iter):
    for w in range(2): qml.Hadamard(wires=w)
    for _ in range(n_iter):
        grover_oracle(best_action, 2)
        diffusion(2)
    return qml.probs(wires=range(2))
